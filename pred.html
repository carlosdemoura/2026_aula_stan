4.1 Predições no STAN

Bayesian models are generative, meaning they can be used to simulate observations just as well as they can be used to estimate parameters. One benefit of this fact is that a statistical model can be developed by thinking hard about how the data might have arisen.”

— Richard McElreath, Statistical Rethinking: A Bayesian Course with Examples in R and Stan (2020)

Código 
mod_FA_pred = stan_model(model_name = "FA_pred", model_code = "
data {
  int<lower=1> n_row;
  int<lower=1> n_col;
  int<lower=1> n_fac;
  matrix[n_row, n_col] X;
  int<lower=1> n_alpha_free;
  real<lower=0> sigma_shape;
  real<lower=0> sigma_rate;
  int<lower=0> alpha_status[n_row, n_fac];
  int<lower=0> n_pred;
  int<lower=0> is_missing[n_row, n_col];
}
parameters {
  vector[n_alpha_free] alpha_;
  matrix[n_fac, n_col] lambda;
  matrix<lower=0>[n_row,1] sigma;
  vector[n_pred] pred_;
}
transformed parameters {
  matrix[n_row, n_fac] alpha;
  matrix[n_row, n_col] pred;
  
  for (i in 1:n_row) { for (k in 1:n_fac) {
    if (alpha_status[i,k] == 0) {
      alpha[i,k] = 0;
    } else {
      alpha[i,k] = alpha_[alpha_status[i,k]];
    }
  }}
  
  for (i in 1:n_row) { for (j in 1:n_col) {
    if (is_missing[i,j] == 0) {
      pred[i,j] = 0;
    } else {
      pred[i,j] = pred_[is_missing[i,j]];
    }
  }}
}
model {
  matrix[n_row, n_col] alpha_lambda;
  alpha_lambda = alpha * lambda;
  for(i in 1:n_row){ for (j in 1:n_col) {
    if (is_missing[i,j] == 0) {
      X[i,j] ~ normal( alpha_lambda[i,j], sigma[i,1] );
    } else {
      pred[i,j] ~ normal( alpha_lambda[i,j], sigma[i,1] );
    }
  }}
  for(i in 1:n_alpha_free) {
    alpha_[i] ~ normal(0,10);
  }
  for (i in 1:n_fac) { for(j in 1:n_col){ 
    lambda[i,j] ~ normal(0,1);
  }}
  for(i in 1:n_row){
    sigma[i,1] ~ gamma(sigma_shape, sigma_rate);
  }
}

")

Código 
group.sizes =
  temperatura |>
  pull(estado) |>
  table()

n = sum(group.sizes)
n.fac = 3

group.limits =
  group.sizes |>
  {\(x) list(
    c(1, (cumsum(x) + 1)[1:(length(x)-1)])[1:length(x)],
    cumsum(x)
  )}()

contr = matrix(0, nrow = n, ncol = n.fac)
for (i in 1:n.fac) {
  contr[group.limits[[1]][i] : group.limits[[2]][i], i] = rep(1, group.sizes[i])
}

contr[contr!=0] = 1:sum(contr!=0)

X =
  temperatura |>
  select(starts_with("2"))

data_FA_pred = list(
  X = X,
  n_row = nrow(X),
  n_col = ncol(X),
  n_fac = 3,
  n_alpha_free = max(contr),
  sigma_shape = .1,
  sigma_rate  = .1,
  alpha_status = contr,
  is_missing = is.missing,
  n_pred = n.pred
)

init = list()
n_chains = 1
for (i in 1:n_chains) {
  init[[i]] = list(
    alpha_ = rep(10, data_FA_pred$n_alpha_free),
    lambda = matrix(0, data_FA_pred$n_fac, data_FA_pred$n_col),
    sigma  = matrix(1, data_FA_pred$n_row, 1)
  )
}
fit_FA_pred = rstan::sampling(mod_FA_pred, data = data_FA_pred, init = init, chains = n_chains)

pred_post =
  rstan::extract(fit_FA_pred, pars = "pred_") |>
  purrr::pluck("pred_")

pred_mean = apply(pred_post, 2, mean)
hpd = coda::HPDinterval(coda::as.mcmc(pred_post), prob = 0.95)
idx_missing = which(is.missing != 0, arr.ind = TRUE)
id_pred = is.missing[idx_missing]

cv =
  tibble(
    linha = idx_missing[,1],
    estado  = temperatura$estado[linha],
    estacao = temperatura$estacao[linha],
    mes   = idx_missing[,2],
    id_pred = id_pred,
    valor_real = as.matrix(X)[idx_missing],
    media_post = pred_mean[id_pred],
    vr = (valor_real - media_post)/abs(valor_real),
    hpd_inf = hpd[id_pred, 1],
    hpd_sup = hpd[id_pred, 2],
    hpd_contem = (valor_real >= hpd_inf) & (valor_real <= hpd_sup),
    hpd_amp = hpd_sup - hpd_inf
  ) |>
  select(-c(linha, id_pred))
